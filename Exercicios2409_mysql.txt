-- 1. Banco de dados (troque "rgmdecadaum" pelo seu RGM)
create database ads_rgmdecadaum;


-- 2. Tabelas 

CREATE TABLE vendas (
    id          INT AUTO_INCREMENT PRIMARY KEY,
    produto     VARCHAR(80) NOT NULL,
    quantidade  INT,
    valor       DECIMAL(10,2),
    data_venda  DATE,
    id_cliente  INT,
    id_vendedor INT,
    FOREIGN KEY (id_cliente)  REFERENCES cliente (id),
    FOREIGN KEY (id_vendedor) REFERENCES vendedor (id)
);

CREATE TABLE cliente (
    id     INT AUTO_INCREMENT PRIMARY KEY,
    nome   VARCHAR(120) NOT NULL,
    cidade VARCHAR(60),
    email  VARCHAR(120)
);

CREATE TABLE vendedor (
    id    INT AUTO_INCREMENT PRIMARY KEY,
    nome  VARCHAR(120) NOT NULL,
    setor VARCHAR(40)
);
